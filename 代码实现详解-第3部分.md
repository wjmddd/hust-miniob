## 10. select-tables - 多表笛卡尔积

### 📝 功能概述

多表查询（笛卡尔积）：
```sql
SELECT * FROM t1, t2;              -- 笛卡尔积
SELECT * FROM t1, t2 WHERE t1.id = t2.id;  -- 带条件的笛卡尔积
```

结果：t1每一行与t2的每一行组合

### 🗂️ 关键文件

```
Resolver: select_stmt.cpp (多表处理)
Operator: join_physical_operator.cpp (笛卡尔积实现)
```

```cpp
// 在 LogicalPlanGenerator::create_plan() 中
RC LogicalPlanGenerator::create_plan(SelectStmt *select_stmt,
                                      unique_ptr<LogicalOperator> &logical_operator)
{
  const std::vector<Table *> &tables = select_stmt->tables();
  
  // ← 断点：检查表数量
  if (tables.size() == 1) {
    // 单表查询
    return create_single_table_plan(select_stmt, logical_operator);
  }
  
  // ← 断点：多表查询，创建笛卡尔积
  LOG_DEBUG("creating cartesian product for %zu tables", tables.size());
  
  unique_ptr<LogicalOperator> cartesian_product;
  
  // 2.1 创建第一个表的扫描
  auto table_get_oper = make_unique<TableGetLogicalOperator>(
      tables[0],  // ← t1
      true        // readonly
  );
  
  cartesian_product = std::move(table_get_oper);
  LOG_DEBUG("base table: %s", tables[0]->name());
  
  // 2.2 依次与其他表做笛卡尔积
  for (size_t i = 1; i < tables.size(); i++) {  // ← 断点：循环
    LOG_DEBUG("joining with table: %s", tables[i]->name());
    
    auto right_table_oper = make_unique<TableGetLogicalOperator>(
        tables[i],  // ← t2
        true
    );
    
    // 创建JOIN算子（不带ON条件 = 笛卡尔积）
    auto join_oper = make_unique<JoinLogicalOperator>();
    join_oper->add_child(std::move(cartesian_product));  // ← 左表
    join_oper->add_child(std::move(right_table_oper));   // ← 右表
    
    cartesian_product = std::move(join_oper);
  }
  
  // ← 断点：查看执行计划树
  // Join
  //   ├─ TableGet(t1)
  //   └─ TableGet(t2)
  
  // 3. 添加过滤条件（WHERE）
  if (select_stmt->filter_stmt() != nullptr) {  // ← 断点
    LOG_DEBUG("adding predicate filter");
    
    auto predicate_oper = make_unique<PredicateLogicalOperator>(
        select_stmt->filter_stmt()->filter_expr()
    );
    predicate_oper->add_child(std::move(cartesian_product));
    logical_operator = std::move(predicate_oper);
  } else {
    logical_operator = std::move(cartesian_product);
  }
  
  return RC::SUCCESS;
}
```

```cpp
RC JoinPhysicalOperator::next()
{
  RC rc = RC::SUCCESS;
  
  // ← 断点：嵌套循环
  while (true) {
    // 1. 如果右表当前元组为空，从左表获取下一行
    if (right_tuple_ == nullptr) {  // ← 断点
      LOG_DEBUG("getting next from left table");
      
      rc = children_[0]->next();  // ← 从t1获取下一行
      if (rc != RC::SUCCESS) {
        LOG_DEBUG("left table exhausted");
        return rc;  // 左表遍历完毕
      }
      
      left_tuple_ = children_[0]->current_tuple();  // ← 获取左元组
      // 例如：left_tuple = (1, 'Alice')
      
      // 重新打开右表（重置扫描）
      LOG_DEBUG("resetting right table scan");
      children_[1]->close();
      children_[1]->open(trx_);
      round_done_ = false;
    }
    
    // 2. 从右表获取下一行
    rc = children_[1]->next();  // ← 从t2获取下一行
    
    if (rc == RC::SUCCESS) {  // ← 断点
      // 右表有数据
      right_tuple_ = children_[1]->current_tuple();  // ← 获取右元组
      // 例如：right_tuple = (1, 25)
      
      LOG_DEBUG("left: %s, right: %s",
                left_tuple_->to_string().c_str(),
                right_tuple_->to_string().c_str());
      
      // 3. 没有JOIN条件 = 笛卡尔积，直接返回组合
      if (join_condition_ == nullptr) {  // ← 断点
        LOG_DEBUG("no join condition, returning cartesian product");
        return RC::SUCCESS;  // ← 返回这一对元组
      }
      
      // 有JOIN条件，计算条件（在JOIN场景中调试）
      // ...
      
    } else if (rc == RC::RECORD_EOF) {  // ← 断点
      // 右表遍历完毕，重置为NULL，继续左表下一行
      LOG_DEBUG("right table exhausted for current left row");
      right_tuple_ = nullptr;
      round_done_ = true;
      // 继续循环，会从左表获取下一行
    } else {
      // 错误
      return rc;
    }
  }
```

```cpp
RC SelectStmt::create(Db *db, SelectSqlNode &select_sql, Stmt *&stmt)
{
  // 1. 处理多个表
  std::vector<Table *> tables;
  std::unordered_map<std::string, Table *> table_map;
  
  for (size_t i = 0; i < select_sql.relations.size(); i++) {
    const char *table_name = select_sql.relations[i].c_str();
    
    Table *table = db->find_table(table_name);
    if (nullptr == table) {
      LOG_WARN("no such table. db=%s, table_name=%s", 
               db->name(), table_name);
      return RC::SCHEMA_TABLE_NOT_EXIST;
    }
    
    tables.push_back(table);
    table_map.insert({table_name, table});
  }

  // 2. 展开字段：SELECT * 需要展开所有表的所有字段
  std::vector<std::unique_ptr<Expression>> query_expressions;
  
  for (const SelectAttr &select_attr : select_sql.attributes) {
    if (select_attr.is_star) {
      // * 的情况
      if (select_attr.relation_name.empty()) {
        // SELECT * FROM t1, t2 → 展开为 t1.*, t2.*
        for (Table *table : tables) {
          for (int i = table->table_meta().sys_field_num(); 
               i < table->table_meta().field_num(); i++) {
            const FieldMeta *field = table->table_meta().field(i);
            query_expressions.push_back(
                std::make_unique<FieldExpr>(table, field)
            );
          }
        }
      } else {
        // SELECT t1.* FROM t1, t2 → 只展开t1的字段
        Table *table = table_map[select_attr.relation_name];
        for (int i = table->table_meta().sys_field_num(); 
             i < table->table_meta().field_num(); i++) {
          const FieldMeta *field = table->table_meta().field(i);
          query_expressions.push_back(
              std::make_unique<FieldExpr>(table, field)
          );
        }
      }
    } else {
      // 具体字段
      Expression *expr = nullptr;
      RC rc = create_expression(select_attr, tables, expr);
      if (rc != RC::SUCCESS) {
        return rc;
      }
      query_expressions.push_back(std::unique_ptr<Expression>(expr));
    }
  }

  // 3. 处理WHERE条件（可能涉及多表）
  FilterStmt *filter_stmt = nullptr;
  RC rc = FilterStmt::create(
      db, 
      nullptr,                    // default_table
      tables.data(), 
      tables.size(),
      select_sql.conditions, 
      filter_stmt
  );

  // 4. 创建SelectStmt
  SelectStmt *select_stmt = new SelectStmt();
  select_stmt->tables_.swap(tables);
  select_stmt->query_expressions_.swap(query_expressions);
  select_stmt->filter_stmt_ = filter_stmt;
  
  stmt = select_stmt;
  return RC::SUCCESS;
}
```

**多表查询的执行计划生成**：

```cpp
RC LogicalPlanGenerator::create_plan(SelectStmt *select_stmt, 
                                      unique_ptr<LogicalOperator> &logical_operator)
{
  const std::vector<Table *> &tables = select_stmt->tables();
  
  // 1. 如果只有一个表，创建简单的表扫描
  if (tables.size() == 1) {
    return create_single_table_plan(select_stmt, logical_operator);
  }
  
  // 2. 多表查询：创建笛卡尔积
  unique_ptr<LogicalOperator> cartesian_product;
  
  // 2.1 创建第一个表的扫描
  auto table_get_oper = make_unique<TableGetLogicalOperator>(
      tables[0], 
      true  // readonly
  );
  cartesian_product = std::move(table_get_oper);
  
  // 2.2 依次与其他表做笛卡尔积
  for (size_t i = 1; i < tables.size(); i++) {
    auto right_table_oper = make_unique<TableGetLogicalOperator>(
        tables[i], 
        true
    );
    
    // 创建JOIN算子（不带ON条件 = 笛卡尔积）
    auto join_oper = make_unique<JoinLogicalOperator>();
    join_oper->add_child(std::move(cartesian_product));
    join_oper->add_child(std::move(right_table_oper));
    
    cartesian_product = std::move(join_oper);
  }
  
  // 3. 添加过滤条件（WHERE）
  if (select_stmt->filter_stmt() != nullptr) {
    auto predicate_oper = make_unique<PredicateLogicalOperator>(
        select_stmt->filter_stmt()->filter_expr()
    );
    predicate_oper->add_child(std::move(cartesian_product));
    logical_operator = std::move(predicate_oper);
  } else {
    logical_operator = std::move(cartesian_product);
  }
  
  // 4. 添加投影
  auto project_oper = make_unique<ProjectLogicalOperator>(
      std::move(select_stmt->query_expressions())
  );
  project_oper->add_child(std::move(logical_operator));
  logical_operator = std::move(project_oper);
  
  return RC::SUCCESS;
}
```

#### 11.2 嵌套循环连接实现

**join_physical_operator.cpp**：

```cpp
class JoinPhysicalOperator : public PhysicalOperator
{
public:
  JoinPhysicalOperator(unique_ptr<Expression> join_condition)
    : join_condition_(std::move(join_condition))
  {}

  RC open(Trx *trx) override
  {
    if (children_.size() != 2) {
      LOG_WARN("join operator should have 2 children");
      return RC::INTERNAL;
    }

    // 打开左表
    RC rc = children_[0]->open(trx);
    if (rc != RC::SUCCESS) {
      return rc;
    }

    // 打开右表
    rc = children_[1]->open(trx);
    if (rc != RC::SUCCESS) {
      return rc;
    }

    trx_ = trx;
    return RC::SUCCESS;
  }

  RC next() override
  {
    // 嵌套循环连接算法
    RC rc = RC::SUCCESS;
    
    while (true) {
      // 1. 如果右表当前元组为空，从左表获取下一行
      if (right_tuple_ == nullptr) {
        rc = children_[0]->next();
        if (rc != RC::SUCCESS) {
          return rc;  // 左表遍历完毕
        }
        
        left_tuple_ = children_[0]->current_tuple();
        
        // 重新打开右表（重置扫描）
        children_[1]->close();
        children_[1]->open(trx_);
        round_done_ = false;
      }
      
      // 2. 从右表获取下一行
      rc = children_[1]->next();
      
      if (rc == RC::SUCCESS) {
        // 右表有数据
        right_tuple_ = children_[1]->current_tuple();
        
        // 3. 计算JOIN条件
        if (join_condition_) {
          // 组合左右元组
          CompositeTuple composite_tuple;
          composite_tuple.add_tuple(left_tuple_);
          composite_tuple.add_tuple(right_tuple_);
          
          // 计算JOIN条件表达式
          Value result;
          rc = join_condition_->get_value(composite_tuple, result);
          
          if (rc != RC::SUCCESS) {
            return rc;
          }
          
          // 4. 检查条件是否满足
          if (result.get_boolean()) {
            // 条件满足，返回这一对元组
            return RC::SUCCESS;
          }
          // 条件不满足，继续下一行
        } else {
          // 没有JOIN条件 = 笛卡尔积
          return RC::SUCCESS;
        }
        
      } else if (rc == RC::RECORD_EOF) {
        // 右表遍历完毕，重置为NULL，继续左表下一行
        right_tuple_ = nullptr;
        round_done_ = true;
      } else {
        // 错误
        return rc;
      }
    }
    
    return RC::RECORD_EOF;
  }

  Tuple *current_tuple() override
  {
    // 返回组合元组
    composite_tuple_.clear();
    composite_tuple_.add_tuple(left_tuple_);
    composite_tuple_.add_tuple(right_tuple_);
    return &composite_tuple_;
  }

  RC close() override
  {
    RC rc = children_[0]->close();
    if (rc != RC::SUCCESS) {
      return rc;
    }
    
    rc = children_[1]->close();
    return rc;
  }

private:
  Trx                        *trx_ = nullptr;
  unique_ptr<Expression>      join_condition_;  // ON条件
  Tuple                      *left_tuple_  = nullptr;
  Tuple                      *right_tuple_ = nullptr;
  CompositeTuple              composite_tuple_;
  bool                        round_done_ = true;
};
```

**算法流程图**：

```
嵌套循环连接：
for each row in left_table:
    for each row in right_table:
        if join_condition(left_row, right_row):
            output (left_row, right_row)

示例：
left_table: (1, 'A'), (2, 'B'), (3, 'C')
right_table: (1, 100), (2, 200), (3, 300)
ON left.id = right.id

执行过程：
1. left=(1,'A'), right=(1,100) → 1==1? YES → output (1,'A',1,100)
2. left=(1,'A'), right=(2,200) → 1==2? NO
3. left=(1,'A'), right=(3,300) → 1==3? NO
4. left=(2,'B'), right=(1,100) → 2==1? NO
5. left=(2,'B'), right=(2,200) → 2==2? YES → output (2,'B',2,200)
6. left=(2,'B'), right=(3,300) → 2==3? NO
7. left=(3,'C'), right=(1,100) → 3==1? NO
8. left=(3,'C'), right=(2,200) → 3==2? NO
9. left=(3,'C'), right=(3,300) → 3==3? YES → output (3,'C',3,300)
```

```cpp
RC OrderByPhysicalOperator::open(Trx *trx)
{
  if (children_.empty()) {
    return RC::INTERNAL;
  }
  
  // 1. 打开子算子
  PhysicalOperator *child = children_[0].get();
  RC rc = child->open(trx);  // ← 断点
  if (rc != RC::SUCCESS) {
    return rc;
  }
  
  // 2. 收集所有数据到内存（ORDER BY需要全部数据）
  std::vector<std::unique_ptr<Tuple>> tuples;
  
  LOG_DEBUG("collecting all tuples for sorting");
  int tuple_count = 0;
  
  // ← 断点：收集数据循环
  while (RC::SUCCESS == (rc = child->next())) {  // ← 循环断点
    Tuple *tuple = child->current_tuple();
    if (nullptr == tuple) {
      LOG_WARN("failed to get current record");
      return RC::INTERNAL;
    }
    
    // 克隆元组（因为child的tuple会被复用）
    tuples.push_back(std::unique_ptr<Tuple>(tuple->clone()));  // ← 断点
    tuple_count++;
    
    LOG_DEBUG("collected tuple #%d: %s", 
              tuple_count, tuple->to_string().c_str());
    
    // ← Watch: tuples.size()
  }
  
  // ← 断点：收集完毕
  LOG_DEBUG("collected %zu tuples for sorting", tuples.size());
  
  if (RC::RECORD_EOF != rc) {
    LOG_WARN("failed to scan table. rc=%s", strrc(rc));
    return rc;
  }
  
  // 3. 排序
  LOG_DEBUG("sorting tuples...");
  
  // ← 断点：排序开始
  std::sort(tuples.begin(), tuples.end(), 
      [this](const std::unique_ptr<Tuple> &a, 
             const std::unique_ptr<Tuple> &b) {
        return this->compare_tuple(a.get(), b.get()) < 0;  // ← F11进入
      }
  );
  
  // ← 断点：排序完成
  LOG_DEBUG("sorting complete");
  
  // 4. 保存排序后的元组
  tuples_.swap(tuples);
  tuple_iterator_ = tuples_.begin();
  
  // ← 检查排序后的顺序
  int i = 0;
  for (const auto &t : tuples_) {
    LOG_DEBUG("sorted tuple #%d: %s", i++, t->to_string().c_str());
  }
  
  return RC::SUCCESS;
}
```

```cpp
{

  LOG_DEBUG("comparing tuples:\n  left: %s\n  right: %s",
            left->to_string().c_str(), right->to_string().c_str());
  
  // 依次比较每个排序字段
  for (const auto &order_by_unit : order_by_units_) {  // ← 循环断点

    Value right_value;
    
    // 获取排序字段的值
    RC rc = order_by_unit->expr()->get_value(*left, left_value);  // ← F11
    if (rc != RC::SUCCESS) {
      return 0;
    }
    
    rc = order_by_unit->expr()->get_value(*right, right_value);  // ← F11
    if (rc != RC::SUCCESS) {
      return 0;
    }
    
    // ← 断点：比较值
    // 例如：比较年龄字段
    // left_value = 20, right_value = 22
    
    int cmp = left_value.compare(right_value);  // ← F11进入
    // cmp = -2 (20 < 22)
    
    LOG_DEBUG("comparing field %s: %s vs %s, cmp=%d, is_desc=%d",
              order_by_unit->field_name().c_str(),
              left_value.to_string().c_str(),
              right_value.to_string().c_str(),
              cmp,
              order_by_unit->is_desc());
    
    // ← 断点：如果是降序，反转比较结果
    if (order_by_unit->is_desc()) {  // ← F10
      cmp = -cmp;  // 反转
      LOG_DEBUG("reversed for DESC: cmp=%d", cmp);
    }
    
    // ← 断点：如果当前字段不相等，返回比较结果
    if (cmp != 0) {  // ← F10
      LOG_DEBUG("fields differ, returning %d", cmp);
      return cmp;  // ← 返回（不需要比较后续字段）
    }
    
    // 当前字段相等，继续比较下一个字段
    LOG_DEBUG("fields equal, comparing next field");
  }
  
  // ← 断点：所有排序字段都相等
  LOG_DEBUG("all fields equal");
  return 0;
}
```

```cpp
class OrderByPhysicalOperator : public PhysicalOperator
{
public:
  // order_by_units: 排序字段和方向
  OrderByPhysicalOperator(
      std::vector<std::unique_ptr<OrderByUnit>> &&order_by_units)
    : order_by_units_(std::move(order_by_units))
  {}

  RC open(Trx *trx) override
  {
    if (children_.empty()) {
      return RC::INTERNAL;
    }

    // 1. 打开子算子
    PhysicalOperator *child = children_[0].get();
    RC rc = child->open(trx);
    if (rc != RC::SUCCESS) {
      return rc;
    }

    // 2. 收集所有数据到内存（ORDER BY需要全部数据）
    std::vector<std::unique_ptr<Tuple>> tuples;
    
    while (RC::SUCCESS == (rc = child->next())) {
      Tuple *tuple = child->current_tuple();
      if (nullptr == tuple) {
        LOG_WARN("failed to get current record");
        return RC::INTERNAL;
      }
      
      // 克隆元组（因为child的tuple会被复用）
      tuples.push_back(std::unique_ptr<Tuple>(tuple->clone()));
    }

    if (RC::RECORD_EOF != rc) {
      LOG_WARN("failed to scan table. rc=%s", strrc(rc));
      return rc;
    }

    // 3. 排序
    std::sort(tuples.begin(), tuples.end(), 
        [this](const std::unique_ptr<Tuple> &a, 
               const std::unique_ptr<Tuple> &b) {
          return this->compare_tuple(a.get(), b.get()) < 0;
        }
    );

    // 4. 保存排序后的元组
    tuples_.swap(tuples);
    tuple_iterator_ = tuples_.begin();

    return RC::SUCCESS;
  }

  RC next() override
  {
    // 从已排序的元组列表中获取下一个
    if (tuple_iterator_ == tuples_.end()) {
      return RC::RECORD_EOF;
    }

    current_tuple_ = tuple_iterator_->get();
    ++tuple_iterator_;

    return RC::SUCCESS;
  }

  Tuple *current_tuple() override
  {
    return current_tuple_;
  }

  RC close() override
  {
    tuples_.clear();
    return children_[0]->close();
  }

private:
  // 比较两个元组
  int compare_tuple(const Tuple *left, const Tuple *right)
  {
    // 依次比较每个排序字段
    for (const auto &order_by_unit : order_by_units_) {
      Value left_value;
      Value right_value;
      
      // 获取排序字段的值
      RC rc = order_by_unit->expr()->get_value(*left, left_value);
      if (rc != RC::SUCCESS) {
        return 0;
      }
      
      rc = order_by_unit->expr()->get_value(*right, right_value);
      if (rc != RC::SUCCESS) {
        return 0;
      }
      
      // 比较值
      int cmp = left_value.compare(right_value);
      
      // 如果是降序，反转比较结果
      if (order_by_unit->is_desc()) {
        cmp = -cmp;
      }
      
      // 如果当前字段不相等，返回比较结果
      if (cmp != 0) {
        return cmp;
      }
      
      // 当前字段相等，继续比较下一个字段
    }
    
    // 所有排序字段都相等
    return 0;
  }

private:
  std::vector<std::unique_ptr<OrderByUnit>> order_by_units_;  // 排序字段
  std::vector<std::unique_ptr<Tuple>>       tuples_;          // 排序后的数据
  std::vector<std::unique_ptr<Tuple>>::iterator tuple_iterator_;
  Tuple                                    *current_tuple_ = nullptr;
};
```

```cpp
class HashGroupByPhysicalOperator : public GroupByPhysicalOperator
{
public:
  RC open(Trx *trx) override
  {
    PhysicalOperator &child = *children_[0];

    RC rc = child.open(trx);
    if (rc != RC::SUCCESS) {
      return rc;
    }

    // 1. 遍历所有数据
    while (RC::SUCCESS == (rc = child.next())) {
      Tuple *child_tuple = child.current_tuple();
      
      // 2. 计算GROUP BY字段的值（作为哈希键）
      GroupKey group_key;
      rc = make_group_key(*child_tuple, group_key);
      if (rc != RC::SUCCESS) {
        return rc;
      }
      
      // 3. 查找或创建对应的分组
      GroupType *found_group = nullptr;
      rc = find_or_create_group(group_key, found_group);
      if (rc != RC::SUCCESS) {
        return rc;
      }
      
      // 4. 计算聚合表达式的值
      ExpressionTuple expr_tuple(value_expressions_);
      expr_tuple.set_tuple(child_tuple);
      
      // 5. 累积到该分组的聚合器
      GroupValueType &group_value = get<1>(*found_group);
      rc = aggregate(get<0>(group_value), expr_tuple);
      if (rc != RC::SUCCESS) {
        return rc;
      }
    }

    if (rc != RC::RECORD_EOF) {
      return rc;
    }

    // 6. 计算所有分组的最终聚合结果
    for (GroupType &group : groups_) {
      GroupValueType &group_value = get<1>(group);
      rc = evaluate(group_value);
      if (rc != RC::SUCCESS) {
        return rc;
      }
    }

    current_group_ = groups_.begin();
    return RC::SUCCESS;
  }

  RC next() override
  {
    // 返回下一个分组
    if (current_group_ == groups_.end()) {
      return RC::RECORD_EOF;
    }

    if (!first_emited_) {
      first_emited_ = true;
      return RC::SUCCESS;
    }

    ++current_group_;
    if (current_group_ == groups_.end()) {
      return RC::RECORD_EOF;
    }

    return RC::SUCCESS;
  }

  Tuple *current_tuple() override
  {
    // 返回当前分组的结果元组
    GroupValueType &group_value = get<1>(*current_group_);
    return &get<1>(group_value);
  }

private:
  // 计算分组键
  RC make_group_key(const Tuple &tuple, GroupKey &group_key)
  {
    // 依次计算每个GROUP BY表达式的值
    for (Expression *expr : group_by_exprs_) {
      Value value;
      RC rc = expr->get_value(tuple, value);
      if (rc != RC::SUCCESS) {
        return rc;
      }
      group_key.push_back(value);
    }
    return RC::SUCCESS;
  }

  // 查找或创建分组
  RC find_or_create_group(const GroupKey &group_key, GroupType *&group)
  {
    // 在哈希表中查找
    for (GroupType &g : groups_) {
      const GroupKey &key = get<0>(g);
      if (group_key_equal(key, group_key)) {
        group = &g;
        return RC::SUCCESS;
      }
    }
    
    // 不存在，创建新分组
    GroupType new_group;
    get<0>(new_group) = group_key;  // 分组键
    
    // 创建聚合器
    AggregatorList aggregator_list;
    create_aggregator_list(aggregator_list);
    get<0>(get<1>(new_group)) = std::move(aggregator_list);
    
    groups_.push_back(std::move(new_group));
    group = &groups_.back();
    
    return RC::SUCCESS;
  }

  // 比较两个分组键是否相等
  bool group_key_equal(const GroupKey &k1, const GroupKey &k2)
  {
    if (k1.size() != k2.size()) {
      return false;
    }
    
    for (size_t i = 0; i < k1.size(); i++) {
      if (k1[i].compare(k2[i]) != 0) {
        return false;
      }
    }
    
    return true;
  }

private:
  using GroupKey = std::vector<Value>;  // 分组键
  using GroupType = std::tuple<GroupKey, GroupValueType>;  // (键, 聚合结果)
  
  std::vector<unique_ptr<Expression>> group_by_exprs_;  // GROUP BY表达式
  std::vector<GroupType>              groups_;          // 所有分组
  std::vector<GroupType>::iterator    current_group_;   // 当前分组
  bool                                first_emited_ = false;
};
```

```cpp
class Aggregator
{
public:
  virtual ~Aggregator() = default;

  // 累积一个值
  virtual void accumulate(const Value &value) = 0;
  
  // 计算最终结果
  virtual Value evaluate() = 0;
};

// COUNT聚合器
class CountAggregator : public Aggregator
{
public:
  void accumulate(const Value &value) override
  {
    if (field_name_ == "*") {
      // COUNT(*): 计数所有行
      count_++;
    } else {
      // COUNT(field): 只计数非NULL值
      if (!value.is_null()) {
        count_++;
      }
    }
  }

  Value evaluate() override
  {
    return Value(count_);
  }

private:
  std::string field_name_;
  int         count_ = 0;
};

// SUM聚合器
class SumAggregator : public Aggregator
{
public:
  void accumulate(const Value &value) override
  {
    // 跳过NULL值
    if (value.is_null()) {
      return;
    }

    if (first_) {
      sum_ = value;
      first_ = false;
    } else {
      sum_.add(value);  // 累加
    }
  }

  Value evaluate() override
  {
    if (first_) {
      return Value::null_value();  // 没有有效值
    }
    return sum_;
  }

private:
  Value sum_;
  bool  first_ = true;
};

// AVG聚合器
class AvgAggregator : public Aggregator
{
public:
  void accumulate(const Value &value) override
  {
    // 跳过NULL值
    if (value.is_null()) {
      return;
    }

    if (first_) {
      sum_ = value;
      first_ = false;
    } else {
      sum_.add(value);
    }
    count_++;
  }

  Value evaluate() override
  {
    if (count_ == 0) {
      return Value::null_value();  // 没有有效值
    }
    
    // 计算平均值
    Value avg = sum_;
    avg.divide(Value(count_));
    return avg;
  }

private:
  Value sum_;
  int   count_ = 0;
  bool  first_ = true;
};

      max_ = value;
      first_ = false;
    } else {
      if (value.compare(max_) > 0) {
        max_ = value;  // 更新最大值
      }
    }
```

```cpp
void GroupByPhysicalOperator::create_aggregator_list(AggregatorList &aggregator_list)
{
  // 遍历所有聚合表达式
  for (Expression *expr : aggregate_expressions_) {
    if (expr->type() == ExprType::AGGREGATION) {
      AggregateExpr *agg_expr = static_cast<AggregateExpr *>(expr);
      
      // 根据聚合类型创建对应的聚合器
      switch (agg_expr->aggregate_type()) {
        case AggregateType::COUNT:
          aggregator_list.push_back(make_unique<CountAggregator>(...));
          break;
        case AggregateType::SUM:
          aggregator_list.push_back(make_unique<SumAggregator>(...));
          break;
        case AggregateType::AVG:
          aggregator_list.push_back(make_unique<AvgAggregator>(...));
          break;
        case AggregateType::MAX:
          aggregator_list.push_back(make_unique<MaxAggregator>(...));
          break;
        case AggregateType::MIN:
          aggregator_list.push_back(make_unique<MinAggregator>(...));
          break;
      }
    }
  }
}
```

```cpp
public:
  SubqueryExpression(unique_ptr<PhysicalOperator> physical_operator)
    : physical_operator_(std::move(physical_operator))
  {}

  ExprType type() const override { return ExprType::SUBQUERY; }

  // 获取子查询的值
  RC get_value(const Tuple &tuple, Value &value) override
  {
    // 1. 执行子查询
    RC rc = physical_operator_->open(trx_);
    if (rc != RC::SUCCESS) {
      return rc;
    }

    // 2. 收集子查询结果
    std::vector<Value> results;
    
    while (RC::SUCCESS == (rc = physical_operator_->next())) {
      Tuple *result_tuple = physical_operator_->current_tuple();
      
      // 假设子查询只返回一列
      Value cell_value;
      rc = result_tuple->cell_at(0, cell_value);
      if (rc != RC::SUCCESS) {
        physical_operator_->close();
        return rc;
      }
      
      results.push_back(cell_value);
    }

    physical_operator_->close();

    if (rc != RC::RECORD_EOF) {
      return rc;
    }

    // 3. 设置结果
    if (results.empty()) {
      value.set_null();  // 子查询无结果
    } else if (results.size() == 1) {
      value = results[0];  // 单值子查询
    } else {
      // 多值子查询（用于IN操作）
      value.set_list(results);  // 存储为列表
    }

    return RC::SUCCESS;
  }

private:
  unique_ptr<PhysicalOperator> physical_operator_;  // 子查询的执行计划
  Trx                         *trx_ = nullptr;
};
```

```cpp
RC ComparisonExpr::get_value(const Tuple &tuple, Value &value)
{
  // 1. 计算左操作数
  Value left_value;
  RC rc = left_->get_value(tuple, left_value);
  if (rc != RC::SUCCESS) {
    return rc;
  }

  // 2. 计算右操作数（可能是子查询）
  Value right_value;
  rc = right_->get_value(tuple, right_value);
  if (rc != RC::SUCCESS) {
    return rc;
  }

  // 3. 如果是IN操作
  if (comp_op_ == CompOp::IN || comp_op_ == CompOp::NOT_IN) {
    if (!right_value.is_list()) {
      // 右边不是列表，转换为单元素列表
      std::vector<Value> list = {right_value};
      right_value.set_list(list);
    }

    // 4. 在列表中查找
    const std::vector<Value> &list = right_value.get_list();
    bool found = false;
    
    for (const Value &v : list) {
      if (left_value.compare(v) == 0) {
        found = true;
        break;
      }
    }

    // 5. 返回结果
    if (comp_op_ == CompOp::IN) {
      value.set_boolean(found);
    } else {  // NOT IN
      value.set_boolean(!found);
    }
    
    return RC::SUCCESS;
  }

  // 其他比较操作...
  return compare_value(left_value, right_value, comp_op_, value);
}
```

```cpp
// 在 UpdatePhysicalOperator::open() 中
RC UpdatePhysicalOperator::open(Trx *trx)
{
  if (children_.empty()) {
    return RC::INTERNAL;
  }
  
  // 1. 打开子算子（扫描需要更新的记录）
  PhysicalOperator *child = children_[0].get();
  RC rc = child->open(trx);  // ← 断点
  if (rc != RC::SUCCESS) {
    return rc;
  }
  
  trx_ = trx;
  
  // 2. 收集所有需要更新的记录
  std::vector<Record> old_records;
  
  LOG_DEBUG("collecting records to update");
  
  // ← 断点：收集循环
  while (RC::SUCCESS == (rc = child->next())) {  // ← 循环断点
    Tuple *tuple = child->current_tuple();
    
    // 获取记录
    if (tuple->type() == TupleType::ROW) {  // ← 断点
      RowTuple *row_tuple = static_cast<RowTuple *>(tuple);
      Record record = row_tuple->record();
      
      old_records.push_back(record);  // ← 保存旧记录
      
      LOG_DEBUG("collected record to update: page=%d, slot=%d",
                record.rid().page_num, record.rid().slot_num);
    }
  }
  
  child->close();
  
  // ← 断点：检查收集到的记录数
  LOG_DEBUG("collected %zu records to update", old_records.size());
  
  if (rc != RC::RECORD_EOF) {
    return rc;
  }
  
  // 3. 更新每条记录
  for (const Record &old_record : old_records) {  // ← 循环断点
    LOG_DEBUG("updating record: page=%d, slot=%d",
              old_record.rid().page_num, old_record.rid().slot_num);
    
    // 3.1 创建新记录
    Record new_record;
    rc = make_updated_record(old_record, new_record);  // ← F11进入
    if (rc != RC::SUCCESS) {
      LOG_ERROR("failed to make updated record");
      return rc;
    }
    
    // ← 断点：检查新旧记录
    // old_record: (1, 'Alice', 25)
    // new_record: (1, 'Alice', 26)
    
    // 3.2 删除旧记录（通过事务）
    LOG_DEBUG("deleting old record");
    rc = trx_->delete_record(table_, old_record);  // ← F11
    if (rc != RC::SUCCESS) {
      LOG_ERROR("failed to delete old record");
      return rc;
    }
    
    // 3.3 插入新记录（通过事务）
    LOG_DEBUG("inserting new record");
    rc = trx_->insert_record(table_, new_record);  // ← F11
    if (rc != RC::SUCCESS) {
      LOG_ERROR("failed to insert new record");
      
      // ← 断点：回滚 - 重新插入旧记录
      LOG_DEBUG("rolling back: reinserting old record");
      trx_->insert_record(table_, old_record);
      return rc;
    }
    
    updated_count_++;  // ← 计数
    LOG_DEBUG("updated record #%d", updated_count_);
  }
  
  // ← 断点：更新完成
  LOG_INFO("updated %d records", updated_count_);
  
  return RC::SUCCESS;
}
```

```cpp
                                                 Record &new_record)
{
  // 1. 复制旧记录的数据
  const int record_size = table_->table_meta().record_size();
  char *record_data = new char[record_size];  // ← 断点
  
  // ← 断点：复制整个记录
  memcpy(record_data, old_record.data(), record_size);
  
  // 在内存视图中查看 record_data
  // 应该与 old_record.data() 完全相同
  
  // 2. 更新指定字段的值
  const char *new_value_data = value_.data();  // ← 新值
  int new_value_len = field_meta_->len();
  
  LOG_DEBUG("updating field %s: offset=%d, len=%d",
            field_meta_->name(), field_meta_->offset(), new_value_len);
  
  // ← 断点：覆盖字段数据
  memcpy(record_data + field_meta_->offset(),   // 字段在记录中的位置
         new_value_data,                         // 新值数据
         new_value_len);                         // 字段长度
  
  // ← 断点：检查更新后的内存
  // 在内存视图中查看 record_data + field_meta_->offset()

  // 例如：更新 age 字段从 25 到 26

  // new: ... [0x1A, 0x00, 0x00, 0x00] ...  (26)
  
  // 3. 创建新记录对象
  new_record.set_data_owner(record_data, record_size);  // ← 断点
  
  LOG_DEBUG("created updated record");
```

**update_physical_operator.cpp**：

```cpp
class UpdatePhysicalOperator : public PhysicalOperator
{
public:
  UpdatePhysicalOperator(Table *table, Value value, const FieldMeta *field_meta)
    : table_(table), value_(std::move(value)), field_meta_(field_meta)
  {}

  RC open(Trx *trx) override
  {
    if (children_.empty()) {
      return RC::INTERNAL;
    }

    // 1. 打开子算子（扫描需要更新的记录）
    PhysicalOperator *child = children_[0].get();
    RC rc = child->open(trx);
    if (rc != RC::SUCCESS) {
      return rc;
    }

    trx_ = trx;

    // 2. 收集所有需要更新的记录
    std::vector<Record> old_records;
    
    while (RC::SUCCESS == (rc = child->next())) {
      Tuple *tuple = child->current_tuple();
      
      // 获取记录
      if (tuple->type() == TupleType::ROW) {
        RowTuple *row_tuple = static_cast<RowTuple *>(tuple);
        old_records.push_back(row_tuple->record());
      }
    }

    child->close();

    if (rc != RC::RECORD_EOF) {
      return rc;
    }

    // 3. 更新每条记录
    for (const Record &old_record : old_records) {
      // 3.1 创建新记录
      Record new_record;
      rc = make_updated_record(old_record, new_record);
      if (rc != RC::SUCCESS) {
        return rc;
      }

      // 3.2 删除旧记录（通过事务）
      rc = trx_->delete_record(table_, old_record);
      if (rc != RC::SUCCESS) {
        return rc;
      }

      // 3.3 插入新记录（通过事务）
      rc = trx_->insert_record(table_, new_record);
      if (rc != RC::SUCCESS) {
        // 回滚：重新插入旧记录
        trx_->insert_record(table_, old_record);
        return rc;
      }

      updated_count_++;
    }

    return RC::SUCCESS;
  }

  RC next() override
  {
    return RC::RECORD_EOF;  // UPDATE不返回数据
  }

  RC close() override
  {
    return RC::SUCCESS;
  }

  int updated_count() const { return updated_count_; }

private:
  // 创建更新后的记录
  RC make_updated_record(const Record &old_record, Record &new_record)
  {
    // 1. 复制旧记录的数据
    const int record_size = table_->table_meta().record_size();
    char *record_data = new char[record_size];
    memcpy(record_data, old_record.data(), record_size);

    // 2. 更新指定字段的值
    const char *new_value_data = value_.data();
    int new_value_len = field_meta_->len();
    
    memcpy(record_data + field_meta_->offset(), 
           new_value_data, 
           new_value_len);

    // 3. 创建新记录对象
    new_record.set_data_owner(record_data, record_size);

    return RC::SUCCESS;
  }

private:
  Table            *table_;
  Value             value_;        // 新值
  const FieldMeta  *field_meta_;   // 要更新的字段
  Trx              *trx_ = nullptr;
  int               updated_count_ = 0;
};
```

**关键点**：
1. 先收集所有需要更新的记录
2. 逐个删除旧记录、插入新记录
3. 通过事务管理，保证原子性
4. 索引会自动更新（在insert_record和delete_record中处理）
