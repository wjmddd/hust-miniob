#### 5.2 日期类型实现

**date_type.cpp**：

```cpp
class DateType : public DataType
{
public:
  // 解析日期字符串
  static RC parse_date(const char *s, int &date_value)
  {
    int year, month, day;
    
    // 1. 解析年月日
    if (sscanf(s, "%d-%d-%d", &year, &month, &day) != 3) {
      LOG_WARN("invalid date format: %s", s);
      return RC::INVALID_ARGUMENT;
    }

    // 2. 范围检查
    if (year < 1970 || year > 2038) {
      return RC::INVALID_ARGUMENT;
    }

    // 3. 月份检查
    if (month < 1 || month > 12) {
      return RC::INVALID_ARGUMENT;
    }

    // 4. 天数检查（考虑闰年）
    int max_day = days_in_month(year, month);
    if (day < 1 || day > max_day) {
      return RC::INVALID_ARGUMENT;
    }

    // 5. 转换为整数（自1970-01-01以来的天数）
    date_value = date_to_int(year, month, day);
    return RC::SUCCESS;
  }

  // 判断闰年
  static bool is_leap_year(int year)
  {
    return (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0);
  }

  // 获取某月的天数
  static int days_in_month(int year, int month)
  {
    static const int days[] = {0, 31, 28, 31, 30, 31, 30, 
                                31, 31, 30, 31, 30, 31};
    
    if (month == 2 && is_leap_year(year)) {
      return 29;  // 闰年2月
    }
    return days[month];
  }

  // 将日期转换为整数（天数）
  static int date_to_int(int year, int month, int day)
  {
    // 计算从1970-01-01到指定日期的天数
    int days = 0;
    
    // 加上整年的天数
    for (int y = 1970; y < year; y++) {
      days += is_leap_year(y) ? 366 : 365;
    }
    
    // 加上当年已过月份的天数
    for (int m = 1; m < month; m++) {
      days += days_in_month(year, m);
    }
    
    // 加上当月的天数
    days += day;
    
    return days;
  }

  // 将整数转换为日期字符串
  static std::string int_to_date_string(int date_value)
  {
    // 反向计算年月日
    int year = 1970;
    int remaining_days = date_value;
    
    // 计算年份
    while (remaining_days > 365) {
      int days_in_year = is_leap_year(year) ? 366 : 365;
      if (remaining_days >= days_in_year) {
        remaining_days -= days_in_year;
        year++;
      } else {
        break;
      }
    }
    
    // 计算月份和日期
    int month = 1;
    while (remaining_days > days_in_month(year, month)) {
      remaining_days -= days_in_month(year, month);
      month++;
    }
    int day = remaining_days;
    
    // 格式化为字符串
    char buf[32];
    snprintf(buf, sizeof(buf), "%04d-%02d-%02d", year, month, day);
    return std::string(buf);
  }

  // 比较两个日期
  int compare(const Value &left, const Value &right) const override
  {
    int left_date = left.get_int();   // DATE存储为int
    int right_date = right.get_int();
    return left_date - right_date;
  }
};
```

## 6. text - 长文本类型

### 📝 功能概述

支持 TEXT 类型（最大4096字节）：
- 变长存储
- 超过4096字节自动截断
- 需要调整 record_manager 的实现

### 🗂️ 关键文件

```
Storage: record_manager.cpp, table.cpp
Type:    char_type.cpp (TEXT复用CHAR的逻辑)
```

### 💻 代码详解

#### 6.1 变长记录格式

**记录格式设计**：

```
定长字段记录：
[field1(4B)] [field2(4B)] [field3(10B)] ...

变长字段记录（支持TEXT）：
[record_size(4B)] [null_bitmap(NB)] [field1_offset(2B)] [field1_len(2B)] 
[field2_offset(2B)] [field2_len(2B)] ... [field1_data] [field2_data] ...
```

**record_manager.cpp**：

```cpp
// 变长记录的插入
RC RecordFileHandler::insert_variable_record(
    const char *data, 
    int record_size, 
    RID *rid)
{
  // 1. 找到有足够空间的页面
  BufferPoolManager &bpm = BufferPoolManager::instance();
  Frame *frame = nullptr;
  
  // 遍历已有页面，查找空闲空间
  for (PageNum page_num = 1; page_num < file_header_->allocated_pages; page_num++) {
    RC rc = bpm.get_this_page(file_id_, page_num, &frame);
    if (rc != RC::SUCCESS) {
      continue;
    }
    
    // 检查页面是否有足够空间
    RecordPageHeader *page_header = (RecordPageHeader *)frame->data();
    if (page_header->record_capacity - page_header->record_num >= 1 &&
        page_header->free_space_size >= record_size) {
      // 找到合适的页面
      break;
    }
    
    bpm.unpin_page(file_id_, page_num);
    frame = nullptr;
  }

  // 2. 如果没有合适的页面，分配新页面
  if (frame == nullptr) {
    RC rc = allocate_page(&frame);
    if (rc != RC::SUCCESS) {
      return rc;
    }
  }

  // 3. 在页面中插入记录
  char *page_data = frame->data();
  RecordPageHeader *page_header = (RecordPageHeader *)page_data;
  
  // 计算插入位置
  int offset = page_header->free_space_offset;
  
  // 4. 写入记录大小
  *(int *)(page_data + offset) = record_size;
  offset += sizeof(int);
  
  // 5. 写入实际数据
  memcpy(page_data + offset, data, record_size);
  
  // 6. 更新页面元数据
  page_header->record_num++;
  page_header->free_space_offset += sizeof(int) + record_size;
  page_header->free_space_size -= sizeof(int) + record_size;
  
  // 7. 设置RID
  rid->page_num = frame->page_num();
  rid->slot_num = offset;  // 变长记录使用偏移量作为slot_num
  
  // 8. 标记页面为脏页
  bpm.mark_dirty(file_id_, frame->page_num());
  bpm.unpin_page(file_id_, frame->page_num());
  
  return RC::SUCCESS;
}
```

#### 6.2 TEXT字段处理

**table.cpp**：

```cpp
RC Table::make_record(int value_num, const Value *values, Record &record)
{
  // 1. 计算记录总大小
  int record_size = sizeof(int);  // 记录大小字段
  bool has_variable_field = false;
  
  for (int i = 0; i < value_num; i++) {
    const FieldMeta *field = table_meta_.field(i + sys_field_num);
    
    if (field->type() == AttrType::TEXT) {
      has_variable_field = true;
      
      // TEXT字段：大小 = 2字节长度 + 实际数据长度
      const Value &value = values[i];
      int text_len = value.length();
      
      // 截断超过4096字节的数据
      if (text_len > 4096) {
        text_len = 4096;
      }
      
      record_size += sizeof(short) + text_len;
    } else {
      // 定长字段
      record_size += field->len();
    }
  }

  // 2. 分配记录缓冲区
  char *record_data = (char *)malloc(record_size);
  memset(record_data, 0, record_size);
  
  int offset = 0;
  
  // 3. 写入记录大小
  *(int *)(record_data + offset) = record_size;
  offset += sizeof(int);

  // 4. 写入各字段数据
  for (int i = 0; i < value_num; i++) {
    const FieldMeta *field = table_meta_.field(i + sys_field_num);
    const Value &value = values[i];
    
    if (field->type() == AttrType::TEXT) {
      // TEXT字段：先写长度，再写数据
      int text_len = value.length();
      if (text_len > 4096) {
        text_len = 4096;  // 截断
      }
      
      // 写入长度（2字节）
      *(short *)(record_data + offset) = (short)text_len;
      offset += sizeof(short);
      
      // 写入数据
      memcpy(record_data + offset, value.data(), text_len);
      offset += text_len;
      
    } else {
      // 定长字段：直接写入
      memcpy(record_data + offset, value.data(), field->len());
      offset += field->len();
    }
  }

  // 5. 设置记录对象
  record.set_data_owner(record_data, record_size);
  
  return RC::SUCCESS;
}
```

**记录存储（使用NULL位图）**：

```cpp
// 记录格式（支持NULL）：
// [record_header] [null_bitmap] [field_data...]

// null_bitmap: 每个bit表示一个字段是否为NULL
// 例如：8个字段需要1个字节的bitmap

RC Table::make_record_with_null(int value_num, const Value *values, Record &record)
{
  int null_bitmap_size = (value_num + 7) / 8;  // 向上取整
  int record_size = sizeof(int) + null_bitmap_size;
  
  // 计算实际数据大小
  for (int i = 0; i < value_num; i++) {
    if (!values[i].is_null()) {
      const FieldMeta *field = table_meta_.field(i + sys_field_num);
      record_size += field->len();
    }
  }
  
  char *record_data = (char *)malloc(record_size);
  memset(record_data, 0, record_size);
  
  int offset = sizeof(int);
  
  // 设置NULL位图
  char *null_bitmap = record_data + offset;
  offset += null_bitmap_size;
  
  for (int i = 0; i < value_num; i++) {
    if (values[i].is_null()) {
      // 设置对应的bit为1
      int byte_index = i / 8;
      int bit_index = i % 8;
      null_bitmap[byte_index] |= (1 << bit_index);
    } else {
      // 写入实际数据
      const FieldMeta *field = table_meta_.field(i + sys_field_num);
      memcpy(record_data + offset, values[i].data(), field->len());
      offset += field->len();
    }
  }
  
  *(int *)record_data = record_size;
  record.set_data_owner(record_data, record_size);
  
  return RC::SUCCESS;
}
```

```cpp
class SumAggregator : public Aggregator
{
public:
  void accumulate(const Value &value) override
  {
    // 跳过NULL值
    if (value.is_null()) {
      return;  // ← 关键：忽略NULL
    }

    if (first_) {
      value_ = value;

    } else {
      value_.add(value);
    }
  }
};

class CountAggregator : public Aggregator
{
public:
  void accumulate(const Value &value) override
  {
    // COUNT(*) 计数所有行
    if (field_name_ == "*") {
      count_++;
      return;
    }
    
    // COUNT(field) 只计数非NULL值
    if (!value.is_null()) {
      count_++;  // ← 关键：只计数非NULL
    }
  }
};

class AvgAggregator : public Aggregator
{
public:
  void accumulate(const Value &value) override
  {
    // 跳过NULL值
    if (value.is_null()) {
      return;
    }
    
    sum_.add(value);
    count_++;
  }
  
  Value evaluate() override
  {
    if (count_ == 0) {
      return Value::null_value();  // 没有有效值，返回NULL
    }
    return sum_.divide(count_);
  }
};
```

## 8. multi-index - 多列索引

### 📝 功能概述


```cpp
class IndexMeta

public:
  // 初始化多列索引
```

#### 8.3 多列键的比较

**bplus_tree.cpp**：

```cpp
int BplusTree::compare_key(const char *key1, const char *key2)
{
  int offset = 0;
  
  // 依次比较每个字段
  for (const FieldMeta *field_meta : field_metas_) {
    int cmp = 0;
    
    // 根据字段类型进行比较
    switch (field_meta->type()) {
      case AttrType::INTS: {
        int v1 = *(int *)(key1 + offset);
        int v2 = *(int *)(key2 + offset);
        cmp = v1 - v2;
      } break;
      
      case AttrType::CHARS: {
        cmp = strncmp(key1 + offset, key2 + offset, field_meta->len());
      } break;
      
      case AttrType::FLOATS: {
        float v1 = *(float *)(key1 + offset);
        float v2 = *(float *)(key2 + offset);
        cmp = (v1 > v2) ? 1 : ((v1 < v2) ? -1 : 0);
      } break;
      
      case AttrType::DATE: {
        int v1 = *(int *)(key1 + offset);
        int v2 = *(int *)(key2 + offset);
        cmp = v1 - v2;
      } break;
    }
    
    // 如果当前字段不相等，直接返回比较结果
    if (cmp != 0) {
      return cmp;  // ← 前面字段优先
    }
    
    // 当前字段相等，继续比较下一个字段
    offset += field_meta->len();
  }
  
  // 所有字段都相等
  return 0;
}
```

#### 8.4 前缀匹配查询

**查询优化器选择索引**：

```cpp
RC PhysicalPlanGenerator::try_to_use_index(
    Table *table,
    FilterUnit *filter_unit,
    Index *&index)
{
  // 遍历所有索引
  for (Index *idx : table->indexes()) {
    const std::vector<std::string> &index_fields = idx->index_meta().fields();
    
    // 检查查询条件是否匹配索引前缀
    // 例如：索引是(col1, col2, col3)
    //      查询条件有col1 = ? AND col2 = ?
    //      可以使用索引（前缀匹配）
    
    bool can_use = true;
    int matched_fields = 0;
    
    for (const std::string &field_name : index_fields) {
      // 查找查询条件中是否有这个字段
      bool found = false;
      for (FilterUnit *unit : filter_units) {
        if (unit->field_name() == field_name) {
          found = true;
          matched_fields++;
          break;
        }
      }
      
      if (!found) {
        // 前缀中断了
        break;
      }
    }
    
    // 如果匹配了索引的前缀，可以使用
    if (matched_fields > 0) {
      index = idx;
      return RC::SUCCESS;
    }
  }
  
  return RC::NOT_FOUND;
}
```

## 9. unique - 唯一索引

### 📝 功能概述

### 💻 代码详解

**bplus_tree_index.cpp**：

```cpp
RC BplusTreeIndex::insert_entry(const char *record, const RID *rid)
{
  // 1. 构造键
  char *key = make_key(record);
  
  // 2. 如果是唯一索引，先检查键是否已存在
  if (index_meta_.unique()) {
    std::vector<RID> rids;
    RC rc = index_handler_.scan_equal(key, key, rids);
    
    if (rc == RC::SUCCESS && !rids.empty()) {
      // 键已存在
      LOG_WARN("duplicate key in unique index. key=%s", key_to_string(key).c_str());
      delete[] key;
      return RC::DUPLICATE_KEY;  // ← 返回重复键错误
    }
  }
  
  // 3. 插入索引
  RC rc = index_handler_.insert_entry(key, rid);
  
  delete[] key;
  return rc;
}
```

**插入记录时的检查（table.cpp）**：

```cpp
RC Table::insert_record(Record &record)
{
  // 1. 先插入记录到数据文件
  RC rc = record_handler_->insert_record(
      record.data(), 
      record.len(), 
      &record.rid()
  );
  
  if (rc != RC::SUCCESS) {
    return rc;
  }

  // 2. 更新所有索引
  for (Index *index : indexes_) {
    rc = index->insert_entry(record.data(), &record.rid());
    
    if (rc != RC::SUCCESS) {
      // 索引插入失败（可能是唯一键冲突）
      
      // 回滚：删除已插入的记录
      record_handler_->delete_record(&record.rid());
      
      // 回滚：删除已更新的索引
      for (Index *idx : indexes_) {
        if (idx == index) {
          break;  // 当前索引还没插入成功
        }
        idx->delete_entry(record.data(), &record.rid());
      }
      
      return rc;  // ← 返回错误（RC::DUPLICATE_KEY）
    }
  }

  return RC::SUCCESS;
}
```
