#### 1.1 CREATE TABLE 实现

**语法解析（yacc_sql.y）**：

```yacc
create_table_stmt:
    CREATE TABLE ID LBRACE attr_def attr_def_list RBRACE storage_format {
        $$ = new ParsedSqlNode(SCF_CREATE_TABLE);
        CreateTableSqlNode &create_table = $$->create_table;
        create_table.relation_name = $3;
        
        // 处理字段定义列表
        std::vector<AttrInfoSqlNode> *src_attrs = $6;

            create_table.attr_infos.swap(*src_attrs);

        std::reverse(create_table.attr_infos.begin(), 
                     create_table.attr_infos.end());

    }
    ;

    ID type LBRACE number RBRACE nullable {
        $$ = new AttrInfoSqlNode;
        $$->type = (AttrType)$2;
        $$->name = $1;
        $$->length = $4;
        $$->nullable = $6;  // 是否允许NULL
        free($1);

    | ID type nullable {
        $$ = new AttrInfoSqlNode;
        $$->type = (AttrType)$2;
```

#### 1.2 INSERT 实现（单行）

**物理算子（insert_physical_operator.cpp）**：

```cpp
RC InsertPhysicalOperator::open(Trx *trx)
{
  std::vector<Record *> inserted_record;
  
  // 遍历所有要插入的元组
  for (auto value : tuples_) {
    Record record;
    
    // 1. 创建记录（将Value转换为二进制）
    RC rc = table_->make_record(
        static_cast<int>(value.size()),  // 字段数量
        value.data(),                     // 字段值数组
        record                            // 输出：记录对象
    );
    
    if (rc != RC::SUCCESS) {
      LOG_WARN("failed to make record. rc=%s", strrc(rc));
      return rc;
    }
    
    // 2. 通过事务插入记录
    rc = trx->insert_record(table_, record);
    if (rc != RC::SUCCESS) {
      LOG_WARN("failed to insert record by transaction. rc=%s", strrc(rc));
      
      // 3. 失败回滚：删除已插入的记录
      for (Record *rec : inserted_record) {
        trx->delete_record(table_, *rec);
      }
      return rc;
    }
    
    // 4. 记录已插入的记录（用于可能的回滚）
    inserted_record.emplace_back(new Record(record));
  }
  
  // 5. 清理临时记录
  for (Record *&rec : inserted_record) {
    delete rec;
  }
  
  return RC::SUCCESS;
}
```

**表层实现（table.cpp）**：

```cpp
RC Table::make_record(int value_num, const Value *values, Record &record)
{
  // 1. 计算记录大小
  int record_size = table_meta_.record_size();
  char *record_data = (char *)malloc(record_size);
  
  // 2. 初始化记录数据（全部设为0）
  memset(record_data, 0, record_size);

  // 3. 遍历每个字段，将Value转换为二进制格式
  for (int i = 0; i < value_num; i++) {
    const FieldMeta *field = table_meta_.field(i + sys_field_num);
    const Value     &value = values[i];
    
    // 计算字段在记录中的偏移量
    size_t copy_len = field->len();
    size_t data_len = copy_len;
    
    // 处理不同类型的数据
    if (field->type() == AttrType::CHARS) {
      // 字符串类型
      const size_t data_len = value.length();
      if (copy_len > data_len) {
        copy_len = data_len;
      }
    }
    
    // 复制数据到记录缓冲区
    memcpy(record_data + field->offset(), value.data(), copy_len);
  }

  // 4. 设置记录对象
  record.set_data_owner(record_data, record_size);
  
  return RC::SUCCESS;
}

RC Table::insert_record(Record &record)
{
  // 1. 插入到记录文件
  RC rc = record_handler_->insert_record(
      record.data(),   // 记录数据
      record.len(),    // 记录长度
      &record.rid()    // 输出：记录ID (页号+槽位)
  );
  
  if (rc != RC::SUCCESS) {
    return rc;
  }

  // 2. 更新所有索引
  for (Index *index : indexes_) {
    rc = index->insert_entry(record.data(), &record.rid());
    if (rc != RC::SUCCESS) {
      // 索引插入失败，删除记录并返回
      record_handler_->delete_record(&record.rid());
      return rc;
    }
  }

  return RC::SUCCESS;
}
```

#### 1.3 SELECT 实现

**查询流程图**：

```
SELECT * FROM t WHERE id > 1
         ↓
1. Parser: 解析为 SelectSqlNode
         ↓
2. Resolver: 创建 SelectStmt
   - 验证表存在
   - 验证字段存在
   - 构建过滤条件
         ↓
3. Optimizer: 生成执行计划
   ProjectPhysicalOperator          ← 投影
     └─ PredicatePhysicalOperator   ← 过滤 (id > 1)
         └─ TableScanPhysicalOperator ← 表扫描
         ↓
4. Executor: 执行
   - TableScan: 逐行读取
   - Predicate: 过滤不满足条件的行
   - Project: 选择需要的列
         ↓
5. 返回结果
```

**表扫描算子（table_scan_physical_operator.cpp）**：

```cpp
RC TableScanPhysicalOperator::open(Trx *trx)
{
  // 打开记录扫描器
  RC rc = table_->get_record_scanner(
      record_scanner_,     // 输出：扫描器对象
      trx,                 // 事务
      mode_                // 扫描模式
  );
  
  if (rc != RC::SUCCESS) {
    LOG_WARN("failed to open record scanner. rc=%s", strrc(rc));
  }
  return rc;
}

RC TableScanPhysicalOperator::next()
{
  // 从扫描器获取下一条记录
  RC rc = record_scanner_.next(current_record_);
  if (rc != RC::SUCCESS) {
    if (RC::RECORD_EOF != rc) {
      LOG_WARN("failed to scan record. rc=%s", strrc(rc));
    }
    return rc;
  }

  // 设置当前元组
  tuple_.set_record(&current_record_);
  return rc;
}
```

```cpp
RC Table::create_index(Trx *trx, bool unique, const char *index_name,
                       const std::vector<std::string> &field_names)
{
  // 1. 验证字段是否存在
  std::vector<const FieldMeta *> field_metas;
  for (const std::string &field_name : field_names) {
    const FieldMeta *field_meta = table_meta_.field(field_name.c_str());
    if (nullptr == field_meta) {
      LOG_WARN("no such field in table. db=%s, table=%s, field=%s",
          base_dir_.c_str(), name(), field_name.c_str());
      return RC::SCHEMA_FIELD_NOT_EXIST;
    }
    field_metas.push_back(field_meta);
  }

  // 2. 创建索引元数据
  IndexMeta new_index_meta;
  RC rc = new_index_meta.init(index_name, field_metas, unique);
  if (rc != RC::SUCCESS) {
    return rc;
  }

  // 3. 创建B+树索引文件
  BplusTreeIndex *index = new BplusTreeIndex();
  std::string index_file = index_data_file(base_dir_.c_str(), name(), index_name);
  
  rc = index->create(this, index_file.c_str(), new_index_meta);
  if (rc != RC::SUCCESS) {
    delete index;
    return rc;
  }

  // 4. 扫描表，将已有数据插入索引
  RecordFileScanner scanner;
  rc = get_record_scanner(scanner, trx, mode);
  
  Record record;
  while (OB_SUCC(rc = scanner.next(record))) {
    rc = index->insert_entry(record.data(), &record.rid());
    if (rc != RC::SUCCESS) {
      break;
    }
  }

  // 5. 更新表元数据
  table_meta_.add_index(new_index_meta);
  indexes_.push_back(index);

  return rc;
}
```

## 2. insert - 多行插入

### 📝 功能概述

支持一条INSERT语句插入多行数据：
```sql
INSERT INTO t VALUES (1, 'a'), (2, 'b'), (3, 'c');
```

**特点**：
- ✅ 原子性：全部成功或全部失败
- ✅ 支持任意多行
- ✅ 失败自动回滚

### 🗂️ 关键文件

```
Parser:    yacc_sql.y (insert_value_list 规则)
Resolver:  insert_stmt.cpp
Executor:  insert_physical_operator.cpp
```

### 💻 代码详解

#### 2.1 语法解析

**yacc_sql.y**：

```yacc
insert_stmt:
    INSERT INTO ID VALUES insert_value_list {
        $$ = new ParsedSqlNode(SCF_INSERT);
        $$->insertion.relation_name = $3;
        
        // $5 是一个 vector<vector<Value>>
        // 外层vector代表多行，内层vector代表每行的多个字段
        if ($5 != nullptr) {
            $$->insertion.tuples.swap(*$5);
            delete $5;
        }
        free($3);
    }
    ;

// 多行值列表：(1,'a'), (2,'b'), (3,'c')
insert_value_list:
    LBRACE value value_list RBRACE {
        // 第一行
        $$ = new std::vector<std::vector<Value>>;
        std::vector<Value> values;
        
        if ($3 != nullptr) {
            values.swap(*$3);
            delete $3;
        }
        values.emplace_back(*$2);
        delete $2;
        
        std::reverse(values.begin(), values.end());
        $$->emplace_back(values);
    }
    | insert_value_list COMMA LBRACE value value_list RBRACE {
        // 后续行
        std::vector<Value> values;
        
        if ($5 != nullptr) {
            values.swap(*$5);
            delete $5;
        }
        values.emplace_back(*$4);
        delete $4;
        
        std::reverse(values.begin(), values.end());
        $$ = $1;
        $$->emplace_back(values);
    }
    ;
```

#### 2.2 Stmt 验证

**insert_stmt.cpp**：

```cpp
RC InsertStmt::create(Db *db, const InsertSqlNode &inserts, Stmt *&stmt)
{
  const char *table_name = inserts.relation_name.c_str();
  
  // 1. 检查表是否存在
  Table *table = db->find_table(table_name);
  if (nullptr == table) {
    LOG_WARN("no such table. db=%s, table_name=%s", 
             db->name(), table_name);
    return RC::SCHEMA_TABLE_NOT_EXIST;
  }

  // 2. 遍历每个需要插入的元组（多行）
  for (auto &values : inserts.tuples) {
    const int value_num = static_cast<int>(values.size());
    const TableMeta &table_meta = table->table_meta();
    const int field_num = table_meta.field_num() - table_meta.sys_field_num();

    // 3. 检查字段数量
    if (field_num != value_num) {
      LOG_WARN("schema mismatch. value num=%d, field num in schema=%d", 
               value_num, field_num);
      return RC::SCHEMA_FIELD_MISSING;
    }

    // 4. 检查每个字段的类型
    const int sys_field_num = table_meta.sys_field_num();
    for (int i = 0; i < field_num; i++) {
      const FieldMeta *field_meta = table_meta.field(i + sys_field_num);
      const AttrType field_type = field_meta->type();
      const AttrType value_type = values[i].attr_type();
      
      // 类型不匹配，尝试转换
      if (value_type != field_type) {
        Value real_value;
        
        // 5. 特殊处理：NULL值
        if (values[i].is_null()) {
          if (not field_meta->nullable()) {
            return RC::NULL_CANT_INSERT;  // 字段不允许NULL
          }
        } else {
          // 6. 类型转换
          RC rc = Value::cast_to(values[i], field_meta->type(), real_value);
          if (OB_FAIL(rc)) {
            LOG_WARN("failed to cast value. table:%s, field:%s, value:%s",
                table_meta.name(), field_meta->name(), 
                values[i].to_string().c_str());
            return rc;
          }
        }
      }
    }
  }

  // 7. 创建 InsertStmt（包含所有元组）
  stmt = new InsertStmt(table, inserts.tuples);
  return RC::SUCCESS;
}
```

#### 2.3 执行插入（原子性保证）

**insert_physical_operator.cpp**：

```cpp
RC InsertPhysicalOperator::open(Trx *trx)
{
  std::vector<Record *> inserted_record;
  
  // 遍历所有元组（可能是多行）
  for (auto value : tuples_) {
    Record record;
    
    // 1. 创建记录
    RC rc = table_->make_record(
        static_cast<int>(value.size()), 
        value.data(), 
        record
    );
    
    if (rc != RC::SUCCESS) {
      LOG_WARN("failed to make record. rc=%s", strrc(rc));
      return rc;
    }
    
    // 2. 插入记录（通过事务）
    rc = trx->insert_record(table_, record);
    
    if (rc != RC::SUCCESS) {
      LOG_WARN("failed to insert record by transaction. rc=%s", strrc(rc));
      
      // 3. 失败回滚：删除本次批量插入中已成功的记录
      // 这保证了原子性：要么全部成功，要么全部失败
      for (Record *rec : inserted_record) {
        trx->delete_record(table_, *rec);
      }
      return rc;
    }
    
    // 4. 记录已插入的记录（用于可能的回滚）
    inserted_record.emplace_back(new Record(record));
  }
  
  // 5. 全部成功，清理临时记录对象
  for (Record *&rec : inserted_record) {
    delete rec;
  }
  
  return RC::SUCCESS;
}
```

**关键点**：
- 使用 `inserted_record` 向量记录已插入的记录
- 一旦某一行插入失败，立即回滚所有已插入的记录
- 这保证了原子性

## 3. select-meta - 元数据检查

### 📝 功能概述

验证查询中的表名、字段名是否存在，不存在则返回FAILURE。

**示例**：
```sql
SELECT nonexist_col FROM t;        -- FAILURE: 字段不存在
SELECT * FROM nonexist_table;      -- FAILURE: 表不存在
SELECT t1.id FROM t2;              -- FAILURE: 表名不匹配
```

### 🗂️ 关键文件

```
Resolver: select_stmt.cpp (验证逻辑)
```

### 💻 代码详解

**select_stmt.cpp**：

```cpp
RC SelectStmt::create(Db *db, SelectSqlNode &select_sql, Stmt *&stmt)
{
  // 1. 验证表是否存在
  std::vector<Table *> tables;
  for (const std::string &table_name : select_sql.relations) {
    Table *table = db->find_table(table_name.c_str());
    if (nullptr == table) {
      LOG_WARN("no such table. db=%s, table_name=%s", 
               db->name(), table_name.c_str());
      return RC::SCHEMA_TABLE_NOT_EXIST;  // ← 返回错误
    }
    tables.push_back(table);
  }

  // 2. 验证查询字段
  std::vector<std::unique_ptr<Expression>> query_expressions;
  for (const SelectAttr &select_attr : select_sql.attributes) {
    
    // 处理 * 的情况
    if (select_attr.is_star) {
      // ... 展开为所有字段
    } else {
      // 3. 验证字段是否存在
      Expression *expr = nullptr;
      RC rc = create_expression(
          select_attr,      // 字段描述
          tables,           // 表列表
          expr,             // 输出：表达式对象
          default_table     // 默认表
      );
      
      if (rc != RC::SUCCESS) {
        LOG_WARN("failed to create expression. rc=%s", strrc(rc));
        return rc;  // ← 字段不存在，返回错误
      }
      
      query_expressions.emplace_back(expr);
    }
  }

  // 4. 验证 WHERE 条件中的字段
  FilterStmt *filter_stmt = nullptr;
  RC rc = FilterStmt::create(
      db, default_table, tables.data(), tables.size(),
      select_sql.conditions, filter_stmt
  );
  
  if (rc != RC::SUCCESS) {
    LOG_WARN("failed to create filter statement. rc=%s", strrc(rc));
    return rc;  // ← WHERE中的字段不存在
  }

  // 5. 所有验证通过，创建 SelectStmt
  stmt = new SelectStmt();
  return RC::SUCCESS;
}
```

**字段验证（expression_binder.cpp）**：

```cpp
RC ExpressionBinder::bind_field(
    const RelAttrSqlNode &attr, 
    Expression *&expr)
{
  const char *table_name = attr.relation_name.c_str();
  const char *field_name = attr.attribute_name.c_str();

  Table *table = nullptr;
  
  // 1. 查找表
  if (common::is_blank(table_name)) {
    // 没有指定表名，在所有表中查找
    for (Table *t : tables_) {
      const FieldMeta *field_meta = t->table_meta().field(field_name);
      if (field_meta != nullptr) {
        if (table != nullptr) {
          // 歧义：多个表都有这个字段
          LOG_WARN("ambiguous field. field=%s", field_name);
          return RC::SCHEMA_FIELD_NAME_DUPLICATE;
        }
        table = t;
      }
    }
  } else {
    // 指定了表名，直接查找
    table = find_table(table_name);
    if (nullptr == table) {
      LOG_WARN("no such table. table=%s", table_name);
      return RC::SCHEMA_TABLE_NOT_EXIST;  // ← 表不存在
    }
  }

  if (nullptr == table) {
    LOG_WARN("no such field. field=%s", field_name);
    return RC::SCHEMA_FIELD_NOT_EXIST;  // ← 字段不存在
  }

  // 2. 在表中查找字段
  const FieldMeta *field_meta = table->table_meta().field(field_name);
  if (nullptr == field_meta) {
    LOG_WARN("no such field in table. table=%s, field=%s", 
             table->name(), field_name);
    return RC::SCHEMA_FIELD_NOT_EXIST;  // ← 字段不存在
  }

  // 3. 创建字段表达式
  expr = new FieldExpr(table, field_meta);
  return RC::SUCCESS;
}
```

**关键点**：
- 在 Resolver 阶段就进行严格验证
- 不存在的表名或字段名会立即返回错误码
- 错误码传播到上层，最终返回 FAILURE

## 4. drop-table - 删除表

### 📝 功能概述

删除表及其所有关联资源：
- 表数据文件
- 所有索引文件
- 表元数据文件

### 🗂️ 关键文件

```
Executor: drop_table_executor.cpp
Storage:  table.cpp, db.cpp
```

### 💻 代码详解

**drop_table_executor.cpp**：

```cpp
RC DropTableExecutor::execute(SQLStageEvent *sql_event)
{
  Stmt    *stmt    = sql_event->stmt();
  Session *session = sql_event->session_event()->session();
  ASSERT(stmt->type() == StmtType::DROP_TABLE, 
         "drop table executor can not run this command: %d", 
         static_cast<int>(stmt->type()));

  DropTableStmt *drop_table_stmt = static_cast<DropTableStmt *>(stmt);

  const char *table_name = drop_table_stmt->table_name().c_str();
  
  // 从数据库中删除表
  RC rc = session->get_current_db()->drop_table(table_name);

  return rc;
}
```

**db.cpp**：

```cpp
RC Db::drop_table(const char *table_name)
{
  // 1. 查找表
  Table *table = find_table(table_name);
  if (nullptr == table) {
    LOG_WARN("no such table. table_name=%s", table_name);
    return RC::SCHEMA_TABLE_NOT_EXIST;
  }

  // 2. 关闭表（释放资源）
  table->close();

  // 3. 删除表的所有文件
  RC rc = table->destroy();
  if (rc != RC::SUCCESS) {
    LOG_ERROR("failed to destroy table. table_name=%s, rc=%s", 
              table_name, strrc(rc));
    return rc;
  }

  // 4. 从内存中移除表对象
  auto iter = std::find_if(opened_tables_.begin(), opened_tables_.end(),
      [table_name](const Table *t) { 
          return 0 == strcmp(t->name(), table_name); 
      });
  
  if (iter != opened_tables_.end()) {
    opened_tables_.erase(iter);
  }

  // 5. 删除表对象
  delete table;

  LOG_INFO("dropped table. table_name=%s", table_name);
  return RC::SUCCESS;
}
```

**table.cpp**：

```cpp
RC Table::destroy()
{
  // 1. 删除所有索引文件
  for (Index *index : indexes_) {
    // 获取索引文件路径
    std::string index_file = index_data_file(
        base_dir_.c_str(), 
        name(), 
        index->index_meta().name()
    );
    
    // 删除索引文件
    if (0 != ::unlink(index_file.c_str())) {
      LOG_ERROR("failed to remove index file. filename=%s, errmsg=%s", 
                index_file.c_str(), strerror(errno));
      return RC::IOERR_DELETE;
    }
    
    delete index;
  }
  indexes_.clear();

  // 2. 删除数据文件
  std::string data_file = table_data_file(base_dir_.c_str(), name());
  if (0 != ::unlink(data_file.c_str())) {
    LOG_ERROR("failed to remove data file. filename=%s, errmsg=%s", 
              data_file.c_str(), strerror(errno));
    return RC::IOERR_DELETE;
  }

  // 3. 删除元数据文件
  std::string meta_file = table_meta_file(base_dir_.c_str(), name());
  if (0 != ::unlink(meta_file.c_str())) {
    LOG_ERROR("failed to remove meta file. filename=%s, errmsg=%s", 
              meta_file.c_str(), strerror(errno));
    return RC::IOERR_DELETE;
  }

  LOG_INFO("table destroyed. table=%s", name());
  return RC::SUCCESS;
}
```

**关键点**：
- 必须删除表的所有关联文件
- 删除顺序：索引文件 → 数据文件 → 元数据文件
- 删除前需要关闭文件句柄
